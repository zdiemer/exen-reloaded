/* exen.FX natives 103..108 (all static). ABI: see Resource.c.txt; EXimg: see AnimBitmap.c.txt.
 * Each wrapper resolves  dst = img_struct(g.pixIma)  and  src = img_struct(image), and calls an
 * entry of dst->ops->fx (ops+0x40) as  fx[k](src, dst, ...).
 *   8-bit vtable: fx table = 0x456018 (installed lazily by 0x42bb30 via 0x415b60).
 *   1-bit vtable: fx table = *(0x457f48) = 0x457ef0, whose first six entries (0x42d489..0x42d4a2)
 *   are EMPTY functions  =>  every FX effect is a silent no-op on 1-bit (monochrome) profiles.
 *   (4-bit gray is not a separate vtable in gensimu as far as seen; conf medium.)
 * All FX implementations below are 8-bit only, copy raw palette indices (no transparency, no paint
 * mode). Rotazoom and mosaic do NOT call sync_clip (Graphics clip ignored; they clamp to the image);
 * shift/shutter call sync_clip first and honour dst clip (+0x18..+0x24).
 * Confidence: wrappers high; inner algorithms medium/low (sketched, not line-by-line).
 */

/* 103 doRotazoom(Graphics g, Image img, int angle, int zoom, int cx, int cy)V @0x424b70 -> fx[0]=0x415b6a
 *   s = sin(angle), c = cos(angle) via 0x41c972/0x41c956 (exen.Math tables);
 *   z = (zoom + 0x10000) >> 8      (zoom is 16.16 *added* to 1.0, i.e. 0 = 1:1);
 *   builds an inverse-mapping walker (start u,v and du,dv per x/y, 16.16) centred on the dst centre
 *   offset by (cx, cy), over the whole dst image; samples src nearest-neighbour.
 *   If src width and height are powers of two (0x41c928) -> 0x415d19 (coordinates wrap with masks,
 *   i.e. the source tiles infinitely), else 0x415e5d (out-of-range samples skipped). */

/* 104 doMosaic(Graphics g, Image img, int x, int y, int level)V @0x424bfd -> fx[1]=0x415fc6
 *   Requires level >= 16 and |x| <= dst.w, |y| <= dst.h, else nothing. Copies src into dst at (x,y)
 *   (negative x/y clamped to 0, size clamped to dst) with nearest-neighbour sampling whose source
 *   step is quantised so that each source sample is replicated over ~level/16 pixels (16 = plain copy,
 *   bigger = coarser blocks), centred.  conf low on the exact block arithmetic (4.12 fixed point). */

/* 105 doShiftHorizontal(Graphics g, Image img, int x, int y, byte[] offsets)V @0x424c84 -> fx[4]=0x41688d
 *   null offsets -> log "pointer array of offset is null" + throw_internal(0).
 *   For each row r of the (clipped) copy: memcpy src row r into dst row (y+r) shifted horizontally by
 *   (s8)offsets[r] + x, clipped to dst clip.  Pointer and length of the array are passed, but the
 *   implementation reads one byte per row WITHOUT checking the array length (OOB if rows > length). */

/* 106 doShiftVertical(Graphics g, Image img, int x, int y, byte[] offsets)V @0x424d62 -> fx[5]=0x416b75
 *   Same per column (each column shifted vertically by offsets[col] + y).  conf medium */

/* 107 doShutterVertical(Graphics g, Image img, int step, boolean dir)V @0x424e40 -> fx[2]=0x4162d0
 *   Processes the image in groups of 8 columns; 0x4165f5 fills a 64-entry table from (ncols/8, step,
 *   dir) giving, per group, how many columns of the group are revealed; revealed columns are copied
 *   src->dst (columns copied pixel by pixel down the image, size = min(src, dst clip)). Used for
 *   "venetian blind" transitions driven by `step` over successive frames.  conf low */
