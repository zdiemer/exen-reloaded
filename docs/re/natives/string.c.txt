     where pstr = struct { u16 len; u8 b[]; }; +0x20 = backing char[] or 0.
   Array header: +0x08 u16 blocksize (incl. 0x14 header), +0x0c u16 typecode, +0x0e u8 pin count,
     +0x12 u16 count, +0x14 data.  The pstr of a dynamic string is &array->count (array+0x12).
   Literals point straight into class constant data (u16-len strings), with +0x20 == 0.
   Helpers: new_obj(0x99,crc) 0x410067; new_array(n,type,crc) 0x410106

int String_length(String *self) { return self->pstr ? self->pstr->len : 0; }      /* 0x42ac79 */

int String_compareTo(String *self, String *other) {                                 /* 0x42a85e */
    if (!other) { throw(NPE /*0x364a8d65*/); return 1; }
    pstr *a = self->pstr; if (!a) return -1;
    pstr *b = other->pstr;                     /* not null-checked */
    for (int i = 0; i < b->len && i < a->len; i++) {
        if (a->b[i] < b->b[i]) return -1;      /* unsigned bytes */
        if (a->b[i] > b->b[i]) return  1;
    }
    if (b->len < a->len) return -1;            /* NB: inverted relative to Java */
    return b->len > a->len ? 1 : 0;
}

String *String_valueOf(int v) {                                                     /* 0x42acb6 */
    int n = 0, t = v < 0 ? -v : v;
    do { n++; t /= 10; } while (t);
    if (v < 0) n++;
    char_array *a = new_array(n, 0x150, 0x9ec04138);
    itoa_rev(v, a->data);                      /* 0x411490: '0' for 0; INT_MIN yields garbage */
    String *s = new_obj(0x99, 0x7772dde3);
    s->array = clone(a); s->array->pin++; s->pstr = &s->array->count;
    return s;
}

static int sb_reserve(StringBuffer *sb, u16 add) {                                   /* 0x42af2f */
    u16 need = add;
    if (sb->array) {
        need += sb->array->count;
        if (need <= sb->array->blocksize - 0x14) return 1;
    }
    if (need > 0x4000) { throw(StringIndexOutOfBounds /*0xbea5520f*/); return 0; }
    char_array *n = new_array(need, 0x150, 0x9ec04138);   /* exact size, no doubling */
    if (!n) return 0;
    n->pin++;
    if (sb->array) {
        n->count = sb->array->count;
        memcpy(n->data, sb->array->data, sb->array->count);
        array_free(sb->array);
    } else n->count = 0;
    sb->array = n; sb->pstr = &n->count;
    return 1;
}
